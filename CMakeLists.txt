cmake_minimum_required(VERSION 3.10)
project(exploration_reconstruction LANGUAGES C CXX)
add_library(exploration_grid STATIC src/grid_state.cpp src/world.cpp src/game.cpp)
set_target_properties(exploration_grid PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(exploration_grid PUBLIC include)
target_compile_features(exploration_grid PUBLIC cxx_std_11)
target_compile_options(exploration_grid PRIVATE -Wall -Wextra -Werror)
if(ANDROID)
  add_library(native_app_glue STATIC ${ANDROID_NDK}/sources/android/native_app_glue/android_native_app_glue.c)
  set_target_properties(native_app_glue PROPERTIES POSITION_INDEPENDENT_CODE ON)
  target_include_directories(native_app_glue PUBLIC ${ANDROID_NDK}/sources/android/native_app_glue)
  add_library(exploration_app SHARED src/android_main.cpp)
  target_compile_features(exploration_app PRIVATE cxx_std_14)
  target_compile_options(exploration_app PRIVATE -Wall -Wextra)
  target_link_libraries(exploration_app PRIVATE exploration_grid native_app_glue android log EGL GLESv1_CM)
  set_property(TARGET exploration_app APPEND_STRING PROPERTY LINK_FLAGS " -u ANativeActivity_onCreate")
else()
  add_executable(exploration src/main.cpp)
  target_link_libraries(exploration PRIVATE exploration_grid)
  enable_testing()
  add_executable(grid_state_test tests/grid_state_test.cpp)
  target_link_libraries(grid_state_test PRIVATE exploration_grid)
  add_test(NAME grid_state COMMAND grid_state_test)
  add_executable(game_test tests/game_test.cpp)
  target_link_libraries(game_test PRIVATE exploration_grid)
  add_test(NAME game COMMAND game_test)
endif()
